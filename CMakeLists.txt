cmake_minimum_required(VERSION 3.20)
project(midi-scope VERSION 0.1.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
find_package(PkgConfig REQUIRED)
pkg_check_modules(ALSA REQUIRED alsa>=1.2.10)   # the UMP API arrived in 1.2.10 (Debian 13 has 1.2.14)

add_executable(midi-scope src/main.cpp)
target_include_directories(midi-scope PRIVATE third_party ${ALSA_INCLUDE_DIRS})
target_link_libraries(midi-scope PRIVATE ${ALSA_LIBRARIES})
target_compile_options(midi-scope PRIVATE -Wall -Wextra -Wno-unused-parameter -Wno-misleading-indentation)

add_executable(midi-scope-testsrc tools/testsrc.cpp)
target_include_directories(midi-scope-testsrc PRIVATE ${ALSA_INCLUDE_DIRS})
target_link_libraries(midi-scope-testsrc PRIVATE ${ALSA_LIBRARIES})

enable_testing()
add_executable(ump_test tests/ump_test.cpp)
add_test(NAME ump_test COMMAND ump_test)

# The launcher keeps drawing its status bar over the LCD; the small script pauses it while the program runs (packaging/midi-scope).
install(PROGRAMS $<TARGET_FILE:midi-scope> DESTINATION share/APPLaunch/bin RENAME midi-scope.bin)
install(PROGRAMS packaging/midi-scope DESTINATION share/APPLaunch/bin)
install(TARGETS midi-scope-testsrc RUNTIME DESTINATION share/APPLaunch/bin)
install(FILES packaging/midi-scope.desktop DESTINATION share/APPLaunch/applications)
install(FILES packaging/midi-scope.png packaging/midi-scope_80.png packaging/midi-scope_100.png DESTINATION share/APPLaunch/share/images)
install(FILES third_party/font8x8_basic.h DESTINATION share/doc/midi-scope RENAME font8x8_basic.h.PublicDomain)
install(FILES LICENSE DESTINATION share/doc/midi-scope RENAME copyright)

set(CPACK_GENERATOR "DEB")
set(CPACK_PACKAGING_INSTALL_PREFIX "/usr")
set(CPACK_PACKAGE_NAME "midi-scope")
set(CPACK_PACKAGE_VERSION "${PROJECT_VERSION}")
set(CPACK_PACKAGE_CONTACT "Leo Kuroshita <7828838+kurogedelic@users.noreply.github.com>")
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "MIDI monitor for the Cardputer Zero: USB, network and MIDI 2.0 (UMP) ports")
set(CPACK_DEBIAN_PACKAGE_ARCHITECTURE "arm64")
set(CPACK_DEBIAN_PACKAGE_DEPENDS "libc6, libstdc++6, libasound2t64 (>= 1.2.10)")
set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS OFF)
set(CPACK_PACKAGE_FILE_NAME "midi-scope_${PROJECT_VERSION}_arm64")
include(CPack)
